----- CORE COMMANDS -----

pwd                   = show current location
mkdir name            = make new folder 
chmod =x file         = make scripts ececutable
chmod 755 file        = owner full control, others can run
chmod 644 file        = owner read/write, others can run
rm file               = delete file forever  (NO recycle bin)
cat file              = read file contents
./script.sh           = run script in current folder
sudo comamnd          = run as administator
